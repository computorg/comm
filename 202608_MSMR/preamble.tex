% code blocks: light Computo-tinted background, no frame
\usepackage[most]{tcolorbox}
\definecolor{shadecolor}{HTML}{F2F6F9}
\makeatletter
\AtBeginDocument{%
  \@ifundefined{Shaded}{}{%
    \renewenvironment{Shaded}{%
      \begin{tcolorbox}[breakable, boxrule=0pt, frame hidden, sharp corners,
        enhanced, borderline west={2pt}{0pt}{computo!35},
        colback=shadecolor, left=4pt, right=4pt, top=2pt, bottom=2pt]}%
      {\end{tcolorbox}}%
  }%
}
\makeatother
